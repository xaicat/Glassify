cmake_minimum_required(VERSION 3.24)
project(Glassify VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_library(jsoncpp STATIC
    third_party/jsoncpp/src/lib_json/json_reader.cpp
    third_party/jsoncpp/src/lib_json/json_value.cpp
    third_party/jsoncpp/src/lib_json/json_writer.cpp)
target_include_directories(jsoncpp PUBLIC third_party/jsoncpp/include)
add_library(wfx_core STATIC src/core/rules.cpp src/core/config.cpp src/integrations/jsonc.cpp)
target_include_directories(wfx_core PUBLIC src)
target_link_libraries(wfx_core PUBLIC jsoncpp)
if(WIN32)
    enable_language(RC)
    add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    add_library(wfx_windows STATIC src/platform/windows.cpp src/platform/overlays.cpp src/platform/mouse.cpp src/platform/focus.cpp src/platform/taskbar.cpp src/recovery/client.cpp src/integrations/terminal.cpp)
    target_include_directories(wfx_windows PUBLIC src)
    target_link_libraries(wfx_windows PUBLIC wfx_core dwmapi user32 gdi32 shell32 advapi32 comctl32 comdlg32 ole32 uxtheme)
    add_executable(WinFadeXNext WIN32 src/app/main.cpp src/app/controller.cpp src/app/theme.cpp src/app/web_host.cpp)
    target_include_directories(WinFadeXNext SYSTEM PRIVATE third_party/webview2/include)
    target_link_libraries(WinFadeXNext PRIVATE ole32 uuid)
    file(GLOB GLASSIFY_UI_ASSETS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/web/*")
    set_property(TARGET WinFadeXNext APPEND PROPERTY LINK_DEPENDS ${GLASSIFY_UI_ASSETS})
    add_custom_command(TARGET WinFadeXNext POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/webview2/x64/WebView2Loader.dll" "$<TARGET_FILE_DIR:WinFadeXNext>"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:WinFadeXNext>/web"
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:WinFadeXNext>/scripts"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/scripts/setup-taskbar.ps1" "$<TARGET_FILE_DIR:WinFadeXNext>/scripts/setup-taskbar.ps1"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/icon.png" "$<TARGET_FILE_DIR:WinFadeXNext>/web/icon.png")
    add_executable(WinFadeXRecovery WIN32 src/recovery/main.cpp)
    set_target_properties(WinFadeXNext PROPERTIES OUTPUT_NAME Glassify)
    set_target_properties(WinFadeXRecovery PROPERTIES OUTPUT_NAME GlassifyRecovery)
    foreach(app WinFadeXNext WinFadeXRecovery)
        target_sources(${app} PRIVATE resources/app.rc)
        target_include_directories(${app} PRIVATE resources)
        target_link_libraries(${app} PRIVATE wfx_windows)
        if(MSVC)
            target_link_options(${app} PRIVATE /MANIFEST:NO)
        endif()
        if(MINGW)
            target_link_options(${app} PRIVATE -municode -static)
        endif()
    endforeach()
    if(BUILD_TESTING)
        # Explicit manual probe only. Automated suites never modify the real taskbar.
        add_executable(wfx_taskbar_probe tests/taskbar_probe.cpp)
        target_link_libraries(wfx_taskbar_probe PRIVATE wfx_windows)
        add_executable(wfx_performance_probe tests/performance_probe.cpp)
        target_link_libraries(wfx_performance_probe PRIVATE wfx_windows psapi)
        if(MINGW)
            target_link_options(wfx_taskbar_probe PRIVATE -static)
            target_link_options(wfx_performance_probe PRIVATE -static)
        endif()
        add_executable(wfx_windows_tests tests/windows_tests.cpp)
        target_link_libraries(wfx_windows_tests PRIVATE wfx_windows)
        if(MINGW)
            target_link_options(wfx_windows_tests PRIVATE -static)
        endif()
        add_test(NAME windows-transactions COMMAND wfx_windows_tests)
        set_tests_properties(windows-transactions PROPERTIES
            ENVIRONMENT "WINFADEX_DATA_DIR=${CMAKE_BINARY_DIR}/test-data"
            TIMEOUT 90)
        add_executable(wfx_ui_tests tests/ui_tests.cpp)
        target_link_libraries(wfx_ui_tests PRIVATE wfx_windows)
        add_dependencies(wfx_ui_tests WinFadeXNext WinFadeXRecovery)
        if(MINGW)
            target_link_options(wfx_ui_tests PRIVATE -static)
        endif()
        add_test(NAME native-ui-workflow COMMAND wfx_ui_tests)
        set_tests_properties(native-ui-workflow PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
        add_test(NAME styled-ui-workflow COMMAND powershell -NoProfile -STA -ExecutionPolicy Bypass
            -File "${CMAKE_SOURCE_DIR}/scripts/test-web-ui.ps1" -BuildDirectory "$<TARGET_FILE_DIR:WinFadeXNext>")
        set_tests_properties(styled-ui-workflow PROPERTIES TIMEOUT 100 RUN_SERIAL TRUE)
    endif()
    install(TARGETS WinFadeXNext WinFadeXRecovery RUNTIME DESTINATION .)
    install(FILES third_party/webview2/x64/WebView2Loader.dll DESTINATION .)
    install(DIRECTORY web/ DESTINATION web)
    install(FILES icon.png DESTINATION web)
    install(FILES scripts/setup-taskbar.ps1 DESTINATION scripts)
    install(DIRECTORY LICENSES/ DESTINATION LICENSES)
    install(FILES START-HERE.md README.md CONTRIBUTING.md SECURITY.md CHANGELOG.md NOTICE LICENSE icon.png DESTINATION .)
    install(FILES config/config.schema.json DESTINATION config)
    install(FILES third_party/jsoncpp/LICENSE DESTINATION third_party/jsoncpp)
    install(FILES third_party/README.md DESTINATION third_party)
    install(FILES third_party/webview2/NOTICE.txt DESTINATION third_party/webview2)
    install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md" PATTERN "*.png" PATTERN "*.json")
endif()
if(BUILD_TESTING)
    add_executable(wfx_core_tests tests/core_tests.cpp)
    target_link_libraries(wfx_core_tests PRIVATE wfx_core)
    if(MINGW)
        target_link_options(wfx_core_tests PRIVATE -static)
    endif()
    add_test(NAME core-rules-and-jsonc COMMAND wfx_core_tests)
endif()
if(MSVC)
    foreach(target wfx_core wfx_windows WinFadeXNext WinFadeXRecovery)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE
                $<$<COMPILE_LANGUAGE:CXX>:/W4>
                $<$<COMPILE_LANGUAGE:CXX>:/permissive->
                $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
        endif()
    endforeach()
else()
    target_compile_options(wfx_core PRIVATE -Wall -Wextra -Wpedantic)
    if(TARGET wfx_windows)
        target_compile_options(wfx_windows PRIVATE -Wall -Wextra -Wpedantic)
        foreach(app WinFadeXNext WinFadeXRecovery)
            target_compile_options(${app} PRIVATE
                $<$<COMPILE_LANGUAGE:CXX>:-Wall>
                $<$<COMPILE_LANGUAGE:CXX>:-Wextra>
                $<$<COMPILE_LANGUAGE:CXX>:-Wpedantic>)
        endforeach()
    endif()
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "Glassify-${PROJECT_VERSION}-windows-x64")
include(CPack)
